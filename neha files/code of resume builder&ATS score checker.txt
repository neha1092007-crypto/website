/* ================= ATS engine ================= */
const VERBS = new Set("built developed designed created led managed analyzed improved increased reduced implemented launched delivered optimized automated cleaned wrote presented organized coordinated supported researched tested deployed trained streamlined collaborated visualized predicted maintained resolved achieved generated established evaluated".split(" "))
const STOP = new Set("the and for with you your are our will have has this that from they their who what about into over more such than also can able work working team role job like not all any new one use using including other must should may etc strong good years year experience skills skill knowledge ability required requirements preferred looking join help make based related field degree plus familiarity need turn".split(" "))
const wc = (t: string) => (t.trim() ? t.trim().split(/\s+/).length : 0)
const rx = (t: string) => t.replace(/[.*+?^${}()|[\]\\/]/g, "\\$&")
const hasTerm = (text: string, term: string) => new RegExp("(^|[^a-z0-9])" + rx(term.toLowerCase()) + "($|[^a-z0-9])").test(text.toLowerCase())
const toText = (r: Resume) => [...Object.values(r.personal), r.summary, ...r.experience.flatMap(j => [j.title, j.company, ...j.bullets]), ...r.education.flatMap(e => [e.school, e.degree, e.year]), ...r.projects.flatMap(p => [p.name, p.description, ...p.bullets]), r.skills.join(", "), r.certifications.join(", ")].filter(Boolean).join("\n")
function keywordsOf(jd: string, n = 25) {
  const f: Record<string, number> = {}
  ;(jd.toLowerCase().match(/[a-z][a-z+#./-]{1,}/g) || []).forEach(w => { const k = w.replace(/[./-]+$/, ""); if (k.length > 2 && !STOP.has(k)) f[k] = (f[k] || 0) + 1 })
  return Object.entries(f).sort((a, b) => b[1] - a[1] || b[0].length - a[0].length).slice(0, n).map(e => e[0])
}
function analyze(r: Resume, jd: string) {
  const text = toText(r)
  const bullets = [...r.experience.flatMap(j => j.bullets), ...r.projects.flatMap(p => p.bullets)].filter(b => b.trim())
  const w = wc(text), checks: Check[] = []
  const add = (id: string, label: string, passed: boolean, weight: number, fix: string) => checks.push({ id, label, passed, weight, fix })
  add("contact", "Contact information", /\S+@\S+\.\S+/.test(r.personal.email) && r.personal.phone.replace(/\D/g, "").length >= 8, 12, "Add a professional email and a phone number.")
  add("name", "Name and location", !!r.personal.name.trim() && !!r.personal.location.trim(), 4, "Add your full name and city.")
  const sw = wc(r.summary)
  add("summary", "Summary of 30-90 words", sw >= 30 && sw <= 90, 8, sw < 30 ? "Write a 2-3 sentence summary of your real background." : "Shorten your summary to under 90 words.")
  add("exp", "Experience or projects listed", r.experience.length + r.projects.length > 0, 14, "Add internships, jobs or projects you have actually done.")
  add("edu", "Education section", r.education.length > 0, 6, "Add your degree and school.")
  add("skills", "Skills section (6+ skills)", r.skills.length >= 6, 10, "List at least 6 skills you genuinely have.")
  const verbOk = bullets.length > 0 && bullets.filter(b => VERBS.has(b.trim().split(/\s+/)[0].toLowerCase().replace(/[^a-z]/g, ""))).length / bullets.length >= 0.6
  add("verbs", "Bullets start with action verbs", verbOk, 10, 'Start bullets with verbs like "Built", "Analyzed", "Led".')
  add("metrics", "Measurable achievements", bullets.filter(b => /\d/.test(b)).length >= 2, 12, "Where true, add numbers: sizes, percentages, time saved, team size.")
  add("blen", "Concise bullets (under 30 words)", bullets.length > 0 && bullets.every(b => wc(b) <= 30), 6, "Split long bullets into shorter ones.")
  add("len", "Length of 250-800 words", w >= 250 && w <= 800, 8, w < 250 ? "Add detail about real projects and responsibilities." : "Trim to one or two pages.")
  let kw: { matched: string[]; missing: string[]; percent: number } | null = null
  if (jd.trim().length > 40) {
    const ks = keywordsOf(jd), matched = ks.filter(k => hasTerm(text, k)), missing = ks.filter(k => !hasTerm(text, k))
    kw = { matched, missing, percent: ks.length ? Math.round((100 * matched.length) / ks.length) : 0 }
    add("kw", "Keyword match with job (50%+)", kw.percent >= 50, 16, "Use the job's wording for skills you genuinely have.")
  }
  const total = checks.reduce((s, c) => s + c.weight, 0)
  const score = Math.round((100 * checks.filter(c => c.passed).reduce((s, c) => s + c.weight, 0)) / total)
  const tips = checks.filter(c => !c.passed).sort((a, b) => b.weight - a.weight).map(c => ({ p: c.weight >= 12 ? "High" : c.weight >= 8 ? "Medium" : "Low", t: c.fix }))
  if (kw?.missing.length) tips.push({ p: "Medium", t: `Job keywords not found: ${kw.missing.slice(0, 6).join(", ")}. Add them only if they honestly describe your experience.` })
  return { score, verdict: score >= 80 ? "Excellent" : score >= 60 ? "Good" : "Needs work", checks, kw, tips }
}

/* ================= Small UI pieces ================= */
const inp = "w-full rounded-xl border border-slate-200 bg-white px-3 py-2 text-sm outline-none focus:ring-2 focus:ring-indigo-500"
const Card = ({ title, children }: { title: string; children: ReactNode }) => (
  <section className="rounded-2xl border border-slate-200 bg-white p-4 shadow-sm"><h3 className="mb-3 text-sm font-semibold text-slate-800">{title}</h3>{children}</section>
)
function Tags({ items, onChange, ph }: { items: string[]; onChange: (v: string[]) => void; ph: string }) {
  const [v, setV] = useState("")
  const add = () => { const t = v.trim(); if (t && !items.some(i => i.toLowerCase() === t.toLowerCase())) onChange([...items, t]); setV("") }
  return (
    <div>
      <div className="mb-2 flex flex-wrap gap-1.5">
        {items.map(i => (<button key={i} onClick={() => onChange(items.filter(x => x !== i))} className="inline-flex items-center gap-1 rounded-full bg-indigo-50 px-2.5 py-1 text-xs font-medium text-indigo-700 hover:bg-indigo-100" aria-label={`Remove ${i}`}>{i}<X size={12} /></button>))}
      </div>
      <div className="flex gap-2"><input className={inp} value={v} placeholder={ph} onChange={e => setV(e.target.value)} onKeyDown={e => { if (e.key === "Enter") { e.preventDefault(); add() } }} /><button onClick={add} className="rounded-xl bg-indigo-600 px-3 text-sm font-semibold text-white hover:bg-indigo-700">Add</button></div>
    </div>
  )
}
function Bullets({ items, onChange }: { items: string[]; onChange: (v: string[]) => void }) {
  return (
    <div className="space-y-2">
      {items.map((b, i) => (
        <div key={i} className="flex gap-2"><textarea rows={2} className={inp} value={b} placeholder="Start with an action verb, add a real number if you have one" onChange={e => onChange(items.map((x, j) => (j === i ? e.target.value : x)))} /><button aria-label="Remove bullet" onClick={() => onChange(items.filter((_, j) => j !== i))} className="text-slate-400 hover:text-red-600"><Trash2 size={16} /></button></div>
      ))}
      <button onClick={() => onChange([...items, ""])} className="inline-flex items-center gap-1 text-xs font-semibold text-indigo-600"><Plus size={14} />Add bullet</button>
    </div>
  )
}
function Ring({ value }: { value: number }) {
  const c = 2 * Math.PI * 52
  return (
    <div className="relative mx-auto h-36 w-36" role="img" aria-label={`ATS score ${value} out of 100`}>
      <svg viewBox="0 0 120 120" className="-rotate-90"><circle cx="60" cy="60" r="52" fill="none" stroke="#e2e8f0" strokeWidth="10" /><circle cx="60" cy="60" r="52" fill="none" stroke={value >= 80 ? "#059669" : value >= 60 ? "#4f46e5" : "#d97706"} strokeWidth="10" strokeLinecap="round" strokeDasharray={c} strokeDashoffset={c * (1 - value / 100)} style={{ transition: "stroke-dashoffset .8s ease" }} /></svg>
      <div className="absolute inset-0 flex flex-col items-center justify-center"><span className="text-4xl font-extrabold text-slate-900">{value}</span><span className="text-xs text-slate-500">ATS score</span></div>
    </div>
  )
}

        </div>

        {/* ATS PANEL */}
        <aside className={`no-print space-y-3 ${show("ats")}`}>
          <Card title="ATS score">
            <Ring value={res.score} />
            <p className={`mt-2 text-center font-bold ${res.score >= 80 ? "text-emerald-600" : res.score >= 60 ? "text-indigo-600" : "text-amber-600"}`} aria-live="polite">{res.verdict}</p>
            <p className="mt-2 text-center text-xs text-slate-500">Only add skills and achievements that are true. Never invent them to raise the score.</p>
          </Card>
          <Card title="Job match">
            <input className={`${inp} mb-2`} placeholder="Job title (optional)" value={jt} onChange={e => setJt(e.target.value)} />
            <textarea rows={5} className={inp} placeholder="Paste a job description to check keywords" value={jd} onChange={e => setJd(e.target.value)} />
            {res.kw ? (<div className="mt-3">
              <div className="mb-1 flex justify-between text-sm font-semibold"><span>Keyword match</span><span>{res.kw.percent}%</span></div>
              <div className="h-2 overflow-hidden rounded-full bg-slate-200"><div className="h-full rounded-full bg-indigo-600 transition-all" style={{ width: res.kw.percent + "%" }} /></div>
              <p className="mt-3 text-xs font-semibold text-slate-500">Matched</p><div className="flex flex-wrap gap-1">{res.kw.matched.map(k => <span key={k} className="rounded-full bg-emerald-50 px-2 py-0.5 text-xs font-medium text-emerald-700">{k}</span>)}</div>
              <p className="mt-3 text-xs font-semibold text-slate-500">Missing</p><div className="flex flex-wrap gap-1">{res.kw.missing.map(k => <span key={k} className="rounded-full bg-red-50 px-2 py-0.5 text-xs font-medium text-red-700">{k}</span>)}</div>
            </div>) : <p className="mt-2 text-xs text-slate-500">Paste a job description to see matched and missing keywords.</p>}
          </Card>
          <Card title="Checklist">
            <ul className="space-y-2">{res.checks.map(c => <li key={c.id} className="flex gap-2 text-sm">{c.passed ? <CheckCircle2 className="mt-0.5 shrink-0 text-emerald-600" size={16} /> : <XCircle className="mt-0.5 shrink-0 text-red-500" size={16} />}<span><b className="font-semibold">{c.label}</b>{!c.passed && <span className="block text-xs text-slate-500">{c.fix}</span>}</span></li>)}</ul>
          </Card>
          <Card title="Tips">
            {res.tips.length ? <ul className="space-y-2">{res.tips.map((t, i) => <li key={i} className="text-sm"><span className={`mr-2 rounded-full px-2 py-0.5 text-xs font-semibold ${t.p === "High" ? "bg-red-50 text-red-700" : t.p === "Medium" ? "bg-amber-50 text-amber-700" : "bg-slate-100 text-slate-600"}`}>{t.p}</span>{t.t}</li>)}</ul> : <p className="text-sm text-emerald-700">Nothing to fix. Great work.</p>}
          </Card>
        </aside>
      </div>
      <footer className="no-print py-8 text-center text-sm text-slate-500">ResumeForge — Hackathon Prototype</footer>
    </main>
  )
}

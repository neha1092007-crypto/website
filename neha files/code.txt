 if(miss.length)sug.push({t:'Missing sections',d:'Consider adding: '+miss.join(', ')+'.',p:'Medium'});
 checks.filter(c=>!c.ok).forEach(c=>sug.push({t:c.name,d:c.tip,p:'ATS'}));
 if(reqYears&&haveYears<reqYears)sug.push({t:'Clarify your experience',d:'The role asks for '+reqYears+'+ years. Describe internships, freelance and project work with dates and what you did. Do not stretch your timeline.',p:'High'});
 if(extra.length)sug.push({t:'Highlight your extra strengths',d:'You also list '+extra.slice(0,6).join(', ')+'. Mention them if they support this role.',p:'Low'});
 sug.push({t:'Final check',d:'Only include skills, titles and achievements that are true. Recruiters often verify them in interviews.',p:'Low'});
 const unaddr=[...req].filter(k=>!hasSkill(Rl,k)).map(k=>'Required skill: '+k);
 if(reqYears&&haveYears<reqYears)unaddr.push('Experience: '+reqYears+'+ years requested, about '+haveYears+' found');
 if(jl&&rlv<jl)unaddr.push('Education level not found in resume');
 if(cats.Certifications.length&&!/certif/.test(Rl))unaddr.push('Certification mentioned in job description');
 return{title:meta.title,company:meta.company,overall,skillsPct,kwPct,expPct,eduPct,atsPct,matched,missing,extra,req:[...req],prf:[...prf],kws,kwMatched,kwMissing,reqYears,haveYears,jl,rlv,cats,checks,sug,unaddr,wordCount:wc};
}

function ComparisonView(r,R,J){

}
